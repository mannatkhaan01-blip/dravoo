// --- UPDATED WORKING AI IMAGE RENDERER (Gemini Nano Banana 2 & GPT Image Gen 2) --- async function generateAIStudioImage(type) { const promptInput = document.getElementById(`ai-prompt-${type}`); const genBtn = document.getElementById(`ai-btn-${type}`); const resultDiv = document.getElementById(`ai-result-${type}`); const resultImg = document.getElementById(`ai-img-src-${type}`); const promptText = promptInput.value.trim(); if (!promptText) return alert("Please enter a creative prompt first!"); const originalBtnText = genBtn.innerText; genBtn.innerText = "GENERATING AI IMAGE..."; genBtn.disabled = true; // AI Image Generation Setup (Using Free High-Speed Pollinations Engine) try { let finalPrompt = promptText; let modelEngine = "flux"; // Default high-quality model // Model ke hissab se engineering settings customize karein if (type === 'banana') { // Nano Banana 2 Style: Creative, Artistic, Vibrant Colors finalPrompt = `${promptText}, vibrant colors, highly detailed digital art, cinematic lighting, masterpiece`; } else if (type === 'gpt') { // GPT Image Gen 2 Style: Photorealistic, Ultra-HD, Sharp Asset finalPrompt = `${promptText}, photorealistic, 4k resolution, hyper-detailed, studio lighting, sharp focus`; } // Ek unique seed generate karte hain taaki har baar nayi image bane const randomSeed = Math.floor(Math.random() * 999999); // Final API URL construct karte hain const apiUrl = `https://image.pollinations.ai/p/${encodeURIComponent(finalPrompt)}?width=600&height=600&seed=${randomSeed}&model=${modelEngine}&nologo=true`; // Image ko client-side par fetch aur load karte hain const response = await fetch(apiUrl); if (!response.ok) throw new Error("AI Generation Engine Timeout."); // Image blob ko preview source mein set karte hain const blob = await response.blob(); resultImg.src = URL.createObjectURL(blob); resultDiv.classList.remove('hidden'); } catch (error) { console.error(error); // Agar API down ho toh backup system call hoga loadFallbackImage(promptText, resultImg, resultDiv); } finally { // UI elements ko wapas normal state me laate hain genBtn.innerText = originalBtnText; genBtn.disabled = false; } } // Backup Fallback System (Agar primary generator fail ho jaye) function loadFallbackImage(prompt, imgElement, divElement) { const cleanPrompt = prompt.replace(/[^\w\s]/gi, ''); const words = cleanPrompt.split(' ').filter(w => w.length > 2); const searchTerms = words.length > 0 ? words.slice(0, 4).join(',') : "abstract,ai"; const uniqueRandomId = Math.floor(Math.random() * 50000); imgElement.src = `https://images.unsplash.com/featured/600x600/?${encodeURIComponent(searchTerms)}&sig=${uniqueRandomId}`; divElement.classList.remove('hidden'); }
